# Runtime dependencies for app.py only -- what a deployed server actually
# imports to answer a request. Verified by tracing app.py's real import graph
# (including what gets unpickled out of artifacts/model.joblib: games_in_use
# and item_feat_df are genuine pandas DataFrames, user_enc/item_enc are
# sklearn LabelEncoders) rather than assumed.
#
# matplotlib, seaborn, shap, lime, torch and pytest are NOT here because
# app.py never touches them -- they belong to training, figure generation and
# tests, not serving. Bundling them anyway is what made the Vercel serverless
# function 546MB and over its limit. See requirements-train.txt.
Flask==3.0.3
numpy
pandas
scipy
# Pinned, unlike the rest of this file: artifacts/model.joblib embeds two
# LabelEncoders (user_enc, item_enc) fitted under 1.7.1. joblib.load prints
# InconsistentVersionWarning against anything else, and a newer release can
# fail to unpickle them outright rather than just warn -- confirmed against
# PyPI to still ship cp310-cp313 wheels for Linux/macOS/Windows, so this pin
# won't break the Vercel build. requirements-train.txt inherits this via
# `-r requirements.txt`, so a future `python train.py` retrains under the same
# version rather than quietly drifting the artifact away from this pin again.
# Re-run train.py and bump this together if the model is ever retrained on a
# newer scikit-learn on purpose.
scikit-learn==1.7.1
joblib
tqdm
